I.	System implementation using SQL Server Management Studio
•	Database creation
CREATE DATABASE FitTrackDB
      III.1.  Tables creation
1.	Create user address table
CREATE TABLE UserAddress(
	User_Address_ID INT PRIMARY KEY NOT NULL,
	User_Street VARCHAR(25),
	User_City VARCHAR(25)
);


2.	Create users table

CREATE TABLE Users(
	UserID INT PRIMARY KEY NOT NULL,
	User_Address_ID INT NOT NULL,
	UserFName VARCHAR(10),
	UserSName VARCHAR(10),
	UserStatus VARCHAR(50),
	UserGender VARCHAR(1),
	UserAge INT, 
	UserWeight DECIMAL(5,2),
	UserHeight DECIMAL(5,2), 
	UserMedicalStatus VARCHAR(30),
	UserPhoneNumber VARCHAR(15),
	UserEmail VARCHAR(30),
	EmergencyNumber VARCHAR(15), 
	FOREIGN KEY(User_Address_ID) REFERENCES UserAddress(User_Address_ID)
);

3.	 Create Goals table

CREATE TABLE Goals (
	GoalID INT PRIMARY KEY NOT NULL,
	UserID INT NOT NULL,
	GoalDescription VARCHAR (50),
	GoalStartDate DATE,
	GoalExpireDate DATE, 
	GoalComplexityLevel VARCHAR(10),
	GoalTargetValue DECIMAL(10, 2),
	FOREIGN KEY(UserID) REFERENCES Users(UserID)
);
4.	 Create Exercise table
CREATE TABLE Exercise (
	ExerciseID INT PRIMARY KEY NOT NULL,
	ExerciseType VARCHAR (50),
	CaloriesBurned DECIMAL(10,2),
	Muscles_Used VARCHAR(50), 
	Exercise_Date DATE,
	Exercise_Start_Time TIME,
	Exercise_End_Time TIME,
);

5.	Create Nutrition table
CREATE TABLE Nutrition (
	Nutrition_ID INT PRIMARY KEY NOT NULL,
	UserID INT NOT NULL,
	Regime_Type VARCHAR(30),
	Number_Lunch INT,
	Caloriesconsumed_ DECIMAL(10,2),
	ProteinsConsumed DECIMAL(8,2),
	Consumption_Date DATE, 
	FOREIGN KEY (UserID) REFERENCES Users(UserID)
);

6.	Create Progress table

	
CREATE TABLE Progress (
	ProgressID INT PRIMARY KEY NOT NULL,
	UserID INT NOT NULL,
	GoalID INT NOT NULL,
	Weight_lost VARCHAR(10),
	Week_frequency INT,
	FOREIGN KEY(UserID) REFERENCES Users(UserID),
	FOREIGN KEY(GoalID) REFERENCES Goals(GOALID)
);


7.	 Create Work out details table

CREATE TABLE Workout_details(
	WorkOut_ID INT PRIMARY KEY NOT NULL,
	Number_of_set INT,  
	Number_of_Rep INT
);
8.	Create ProgressWorkout_jonc table

CREATE TABLE ProgressWorkout_jonc(
	ProgressWorkoutID INT PRIMARY KEY NOT NULL,
	ProgressID INT NOT NULL,
	WorkoutID INT NOT NULL,
	FOREIGN KEY(WorkoutID) REFERENCES Workout_details(WorkOut_ID),
	FOREIGN KEY(ProgressID) REFERENCES Progress(ProgressID)
);

9.	 Create Instructor address table

CREATE TABLE Instructor_Adress(
	Instruc_Address_ID INT PRIMARY KEY NOT NULL,
	Instruct_Street VARCHAR(25),
	Instruct_City VARCHAR(25)
);

10.	 Create Instructor table

CREATE TABLE Instructor(
	InstructorID INT PRIMARY KEY NOT NULL,
	UserID INT NOT NULL,
	Instruc_Address_ID INT NOT NULL,
	Instructor_FName VARCHAR(40),
	Instructor_SName VARCHAR(30), 
	Instructor_Phone_Number VARCHAR(50),
	Instructor_Specialization VARCHAR(40),
	Instructor_aviability VARCHAR(30),
	FOREIGN KEY(UserID) REFERENCES Users(UserID),
	FOREIGN KEY(Instruc_Address_ID)REFERENCES Instructor_Adress(Instruc_Address_ID)
);

-----        PERFORM SOME QUERIES ON Exercise Table
--------------------------------------------------------------------------------------
----Select all records in Exercise Table

SELECT * FROM [dbo].[Exercise];


----Select user who burned more than 2000 Kcal
SELECT* FROM [dbo].[Exercise]
WHERE CaloriesBurned>2000;
 
---- Select Exercice Duration and exercice type from Exercice table
SELECT Exercise_Date, ExerciseType FROM [dbo].[Exercise];

---- Select users who did CardioExercice and start their training on 2025-10-11, and  with calory burned greater than 200 and used to train chest 
SELECT*  FROM  [dbo].[Exercise] WHERE  (ExerciseType='CardioExercice' and Exercise_Start_Time='2025-10-11') AND (CaloriesBurned>200 AND Muscles_Used= 'Chest');

 ---- Count the number of users who did Yoga and with a duration activity equals to 1, and  with calory burned greater than 1200 and used to train Biceps
SELECT COUNT (*)
FROM [dbo].[Exercise] WHERE  (ExerciseType='Yoga' and Exercise_Start_Time='13:30:00') AND (CaloriesBurned>1200 AND Muscles_Used= 'Biceps');

 ---- Select users who did Yoga or running as exercises
SELECT* FROM [dbo].[Exercise] WHERE ExerciseType in ('Yoga', 'Running');

--- Sort the exercise table in descending order by Exercise Date
SELECT* FROM [dbo].[Exercise] order by Exercise_Date DESC


-- Filter the exercice table by Exercice Start time in an ascending order and Muscles_Used in alphabetic order
SELECT* FROM [dbo].[Exercise] ORDER BY Exercise_Start_Time ASC, Muscles_Used ASC

--- Select the 5 first rows of our tables
SELECT TOP 5* FROM [dbo].[Exercise]

--- Select people who did start_exercice between 11:00  and 19:00 
SELECT* FROM [dbo].[Exercise] WHERE Exercise_Start_Time BETWEEN '11:00:00' AND '19:00:00';

--- Compute the average of calory burned per hour for user who did exercise between 11:00  and 19:00 

SELECT 
    UsEx.UserID, 
    AVG(Ex.CaloriesBurned) AS AvgCaloriesBurned
FROM 
    Exercise Ex
JOIN 
    User_Exercise_Jonction UsEx ON Ex.ExerciseID = UsEx.ExerciseID
WHERE 
    Ex.Exercise_Start_Time BETWEEN '11:00:00' AND '19:00:00'
GROUP BY 
    UsEx.UserID;

----- Count the number of exercises performed by each user

SELECT 
    Us.UserID,
    Us.UserFName,
    Us.UserSName,
    COUNT(DISTINCT User_Exercise_Jonction.ExerciseID) AS NbrExercisePerformed
FROM Users Us
JOIN User_Exercise_Jonction ON Us.UserID = User_Exercise_Jonction.UserID
GROUP BY Us.UserID, Us.UserFName, Us.UserSName;

--- select the name of the user who performed multiples exercises

SELECT U.UserID, U.UserFName, U.UserSName, COUNT(User_Exercise_Jonction.ExerciseID) AS ExerciseCount
FROM Users AS U
JOIN User_Exercise_Jonction ON U.UserID = User_Exercise_Jonction.UserID
GROUP BY U.UserID, U.UserFName, U.UserSName
HAVING COUNT(User_Exercise_Jonction.ExerciseID) > 1;

--------------------------------------------------------------------------------------
-----        PERFORM SOME ADVANCED QUERIES JOINGING DIFFERENTS TABLES 
--------------------------------------------------------------------------------------
---- Find Users with Specific Goals and Their Instructors
SELECT u.UserFName, u.UserSName, g.GoalDescription, i.Instructor_FName, i.Instructor_SName
FROM Users u
JOIN Goals g ON u.UserID = g.UserID
JOIN Instructor i ON u.UserID = i.UserID
WHERE g.GoalDescription = 'Fat lose';

----Compute the total Calories Burned by Users with a certain Medical Conditions

SELECT u.UserMedicalStatus, SUM(e.CaloriesBurned) AS TotalCaloriesBurned
FROM Users u
JOIN Exercise e ON u.UserID = e.UserID
WHERE u.UserMedicalStatus IN ('Diabetic', 'Heart Disease')
GROUP BY u.UserMedicalStatus;
---- Average Calories Consumed by Users in Different Regimes
SELECT n.Regime_Type, AVG(n.Ccaloriesconsumed_) AS AverageCaloriesConsumed
FROM Nutrition n
GROUP BY n.Regime_Type;

-----List Users and Their Progress in Different Workouts

SELECT u.UserFName, u.UserSName, p.Weight_losed, w.Number_set, w.Number_Rep
FROM Users u
JOIN Progress p ON u.UserID = p.UserID
JOIN Workout_Details w ON p.WorkoutID = w.WorkOut_ID;

----Find Users with the Most Calories Burned
SELECT TOP 3 u.UserFName, u.UserSName, SUM(e.CaloriesBurned) AS TotalCaloriesBurned
FROM Users u
JOIN Exercise e ON u.UserID = e.UserID
GROUP BY u.UserFName, u.UserSName
ORDER BY TotalCaloriesBurned DESC;

---- Group users according to their instructor
SELECT i.Instructor_FName, i.Instructor_SName, u.UserFName, u.UserSName
FROM Instructor i
JOIN Users u ON i.UserID = u.UserID
ORDER BY i.Instructor_FName, i.Instructor_SName;

---Users who met their goals
SELECT u.UserFName, u.UserSName, g.GoalDescription, g.GoalTargetValue, p.Weight_losed
FROM Users u
JOIN Progress p ON u.UserID = p.UserID
JOIN Goals g ON p.GoalID = g.GoalID
WHERE p.Weight_losed >= g.GoalTargetValue;
----Identify the most common exercise
SELECT TOP 2 e.ExerciseType, COUNT(*) AS ExerciseCount
FROM Exercise e
GROUP BY e.ExerciseType
ORDER BY ExerciseCount DESC;

--- Calculate the calories burned by each gender for a specific exercice type

SELECT u.UserGender, e.ExerciseType, SUM(e.CaloriesBurned) AS TotalCaloriesBurned
FROM Users u
JOIN Exercise e ON u.UserID = e.UserID
WHERE e.ExerciseType = 'Yoga'
GROUP BY u.UserGender, e.ExerciseType;

---Compute or filter users and their nutrition according to their medical status
SELECT user.UserFName, user.UserSName, user.UserMedicalStatus, nut.Regime_Type, nut.Caloriesconsumed_
FROM Users user
JOIN Nutrition nut ON user.UserID = nut.UserID
WHERE user.UserMedicalStatus IN ('Diabetic', 'Heart Disease');

---Users with a top workout frequency
SELECT TOP 2 usr.UserFName, usr.UserSName, MAX(prog.Week_frequency) AS HighestWeeklyFrequency
FROM Users usr
JOIN Progress prog ON usr.UserID = prog.UserID
GROUP BY usr.UserFName, usr.UserSName
ORDER BY HighestWeeklyFrequency DESC;

---Users with most calories consumed on a specific date
SELECT TOP 3 u.UserFName, u.UserSName, SUM(n.Caloriesconsumed_) AS TotalCaloriesConsumed
FROM Users u
JOIN Nutrition n ON u.UserID = n.UserID
WHERE n.Consumption_Date = '2025-03-01'
GROUP BY u.UserFName, u.UserSName
ORDER BY TotalCaloriesConsumed DESC;


--- Find the exercise performed by each user last moth 
SELECT 
    U.UserID,
    U.UserFName,
    U.UserSName,
    Exercise.ExerciseType,
    Exercise.Exercise_Date

---- Find users, their trainers names and their area of specialisation
SELECT 
    U.UserID,
    U.UserFName,
    U.UserSName,
    Ins.Instructor_FName,
    Ins.Instructor_Specialization

--------Identify the top 2 most popular exercise 
SELECT TOP 2
    Exer.ExerciseType,
    COUNT(User_Exercise_Jonction.ExerciseID) AS Frequency 
FROM User_Exercise_Jonction
JOIN Exercise Exer ON User_Exercise_Jonction.ExerciseID = Exer.ExerciseID
GROUP BY Exer.ExerciseType
ORDER BY Frequency DESC;

---- Users with their progress and workout information 
SELECT 
    Users.UserID,
